from flask import Flask, render_template
from db import get_connection

app = Flask(__name__)

@app.route("/")
def home():

    db = get_connection()
    cursor = db.cursor()

    # get all problems
    cursor.execute("""
        SELECT problem_id, question_text, final_answer, category_id
        FROM problems
    """)

    problems = cursor.fetchall()

    all_data = []

    for p in problems:

        # get steps
        cursor.execute("""
            SELECT  step_text
            FROM rules
            WHERE category_id=%s
           
        """, (p[3],))

        steps = cursor.fetchall()

        all_data.append({
            "question": p[1],
            "answer": p[2],
            "steps": steps
        })
    cursor.close()
    db.close()

    return render_template("P.html", problems=all_data)

if __name__ == "__main__":
    app.run(debug=True)
